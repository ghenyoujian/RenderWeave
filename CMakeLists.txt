    src/core/contracts.cpp
    src/core/numerics.cpp
    src/core/state_transaction.cpp

    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
set_target_properties(rw_core PROPERTIES
    VERSION ${PROJECT_VERSION} SOVERSION 0
    EXPORT_NAME core
    INSTALL_RPATH "$ORIGIN"

    target_compile_options(rw_core PRIVATE /W4 /fp:strict)
